// DOM-only rendering: provider and finding text is never inserted as HTML.
const $=id=>document.getElementById(id);
const fmt=value=>value?new Date(Number(value)).toLocaleString('en-US',{timeZone:'America/New_York',month:'short',day:'numeric',hour:'numeric',minute:'2-digit'}):'—';
const node=(tag,cls,text)=>{const el=document.createElement(tag);if(cls)el.className=cls;if(text!==undefined)el.textContent=String(text);return el;};
const clear=el=>{while(el.firstChild)el.firstChild.remove();};
const empty=(el,message)=>el.append(node('div','empty',message));
const dayNY=()=>new Intl.DateTimeFormat('en-CA',{timeZone:'America/New_York',year:'numeric',month:'2-digit',day:'2-digit'}).format(new Date());
function findingCard(f) {
  const evidence=f.evidence || {}, card=node('article','finding'),row=node('div','row'),left=node('div');
  left.append(node('div','title',`${f.game_key} · ${String(f.reason_code || '').replaceAll('_',' ')}`));
  left.append(node('div','sub',`${f.status || 'recorded'} · First seen ${fmt(f.first_seen)} · Last seen ${fmt(f.last_seen)} · ${f.failure_layer || 'unknown layer'}`));
  row.append(left,node('span',`badge ${f.severity || 'medium'}`,String(f.severity || 'medium').toUpperCase()));card.append(row);
  card.append(node('div','detail',evidence.detail || 'Review the recorded evidence.'));
  card.append(node('div','evidence',`Observed: ${JSON.stringify(evidence.observed ?? {})} · Expected: ${evidence.expected || 'review'} · Confidence: ${Math.round(Number(f.confidence || 0)*100)}%`));
  return card;
}
function gameCard(g,findings) {
  const e=g.latestEvidence || {},s=e.state || {},teams=(s.teams || []).map(t=>t?.[0] || '?').join(' vs '),
    associated=findings.filter(f=>f.game_key===g.game_key || f.evidence?.affectedGames?.includes(g.game_key));
  const card=node('article','game'),row=node('div','row'),left=node('div');
  left.append(node('div','title',teams || g.game_key));
  left.append(node('div','sub',`${g.sport} · ${g.observations} observations · ${fmt(g.first_seen)} to ${fmt(g.last_seen)}`));
  row.append(left,node('span',associated.length?'badge medium':'badge',associated.length?`${associated.length} finding${associated.length===1?'':'s'}`:'No findings'));
  card.append(row,node('div','detail',`Rating range ${g.min_rating}–${g.peak_rating} · Latest recorded ${s.score ?? '—'} · ${(s.teams || []).map(t=>t?.[2] ?? '—').join('–')}`));
  if(associated.length)card.append(node('div','evidence',associated.map(f=>String(f.reason_code || '').replaceAll('_',' ')).join(' · ')));
  return card;
}
async function refresh() {
  try {
    const day=$('day').value || dayNY(),res=await fetch(`/report?date=${encodeURIComponent(day)}`,{credentials:'same-origin',cache:'no-store'});
    if(res.status===401){location.href='/dashboard';return}if(!res.ok)throw new Error(`Report HTTP ${res.status}`);
    const data=await res.json(),run=data.latestRun || {},post=data.postSlate || {},all=post.findings || [],active=data.activeUnresolvedFindings || [];
    const history=(post.events || []).map(event=>({game_key:event.game_key,reason_code:event.reason_code,
      status:event.event,first_seen:event.at,last_seen:event.at,severity:event.evidence?.severity || 'medium',
      failure_layer:event.evidence?.failureLayer || 'unknown',confidence:event.evidence?.confidence || 0,
      evidence:event.evidence}));
    const byReason=new Map();for(const f of [...all,...history])byReason.set(`${f.game_key}|${f.reason_code}`,f);
    const healthy=Boolean(run.finished_at && !run.error && Date.now()-run.finished_at<8*60000);
    $('status-dot').classList.toggle('warn',!healthy);
    $('status').textContent=run.error?'Audit scan failed':healthy?'Monitoring active':'Waiting for scheduled scan';
    $('updated').textContent=`Last successful audit ${fmt(data.lastSuccessfulScan)} · Feed age ${data.sourceFreshnessMs==null?'—':Math.round(data.sourceFreshnessMs/60000)+' min'}`;
    $('games-checked').textContent=run.games_checked ?? '—';$('checks-completed').textContent=run.checks_completed ?? '—';
    $('clean-games').textContent=run.clean_games ?? '—';$('active-findings').textContent=data.activeUnresolvedCount ?? 0;
    clear($('findings'));if(active.length)active.forEach(f=>$('findings').append(findingCard(f)));else empty($('findings'),'No active anomalies.');
    $('night-summary').textContent=`${post.distinctGames || 0} distinct games · ${post.scans || 0} scheduled scans · ${post.checksCompleted || 0} checks · ${history.filter(f=>f.status==='opened').length} findings opened`;
    clear($('games'));if(post.gameSummaries?.length)post.gameSummaries.forEach(g=>$('games').append(gameCard(g,[...byReason.values()])));else empty($('games'),'No live games have been audited on this night yet. Inactive slates are skipped.');
    clear($('history'));if(history.length)history.forEach(f=>$('history').append(findingCard(f)));else empty($('history'),'No confirmed finding events recorded on this night.');
  } catch(error){$('status-dot').classList.add('warn');$('status').textContent='Dashboard data unavailable';$('updated').textContent=String(error)}
}
async function init(){
  try{const response=await fetch('/export/dates',{credentials:'same-origin',cache:'no-store'});
    if(response.ok){const dates=await response.json();$('day').value=dates.defaultDate || dayNY();}
    else $('day').value=dayNY();
  }catch{$('day').value=dayNY()}
  await refresh();
}
const valueText=value=>value===null||value===undefined?'unavailable':typeof value==='object'?JSON.stringify(value):String(value);
const label=value=>String(value||'insufficient evidence').replaceAll('_',' ');
const trajectoryId=(key,model)=>'trajectory-'+encodeURIComponent(key+'|'+model);
function evidenceDetails(title,value){
  const detail=node('details');detail.append(node('summary',null,title),node('pre','evidence',JSON.stringify(value,null,2)));return detail;
}
function observationLink(key,model,text='Supporting observations'){
  const link=node('a','evidence-link',text);link.href='#'+trajectoryId(key,model);return link;
}
function qualityCard(r){
  const card=node('article','finding');
  card.append(node('div','title',`${r.matchup||r.gameKey} \u00b7 rating ${r.rating??'unavailable'} \u00b7 ${(r.scores||[]).join('-')}`));
  card.append(node('div','sub',`${r.gameKey} \u00b7 ${fmt(r.observedAt)} \u00b7 period ${r.period??'?'}, ${r.clockSeconds??'?'}s \u00b7 ${r.modelVersion}`));
  card.append(node('div','detail',`${label(r.classification)} \u00b7 ${r.confidence||'unknown'} confidence. ${r.suspectedCause||r.whyInvestigate||r.reason||''}`));
  card.append(node('div','evidence',`Evidence: ${(r.signals||[]).join('; ')||valueText(r.verifiedSituation??r.situation)}. Missing: ${(r.missingEvidence||r.evidenceGaps||[]).join('; ')||'none identified'}.`));
  if(r.sourceEvent)card.append(node('div','evidence',`Source event ${r.sourceEvent.id} \u00b7 event time ${fmt(r.sourceEvent.occurredAt)} \u00b7 ${r.sourceEvent.text||''}`));
  card.append(node('div','detail',`Next: ${r.recommendedCorrection||r.proposedInvestigation||r.proposedCorrection||'Replay matched states before changing scoring.'}`));
  card.append(observationLink(r.gameKey,r.modelVersion),evidenceDetails('Investigation evidence and regression risks',r));
  return card;
}
function qualitySection(parent,id,title){
  const section=node('section','quality-section');section.id=id;section.append(node('h3',null,title));parent.append(section);return section;
}
function renderQuality(q,el,date){
  const games=q.games||[],recommendations=q.recommendations||[];
  const navigation=node('nav','quality-nav');navigation.ariaLabel='Scoring quality sections';
  for(const [id,title] of [['quality-summary','Summary'],['quality-priority','Concerns'],['quality-events','Events'],['quality-alerts','Alert usefulness'],['quality-calibration','Calibration'],['quality-evidence','Evidence'],['quality-recommendations','Investigations']]){
    const link=node('a',null,title);link.href='#'+id;navigation.append(link);
  }
  el.append(navigation);
  const overview=qualitySection(el,'quality-summary','Executive summary');
  overview.append(node('p','detail',`${games.length} game/model trajectories; ${(q.modelCohorts||[]).length} separate model cohorts. ${(q.confirmedScoringDefects||[]).length} confirmed scoring defects; ${(q.topOverrated||[]).length} overrating reviews and ${(q.topUnderrated||[]).length} underrating reviews.`));
  overview.append(node('p','evidence','A concern is an investigation, not a scoring change. Completed ratings, live peaks and current watchability have different meanings. One slate cannot establish sport calibration.'));
  if(q.executiveSummary)overview.append(evidenceDetails('Summary evidence',q.executiveSummary));
  const integrity=qualitySection(el,'quality-integrity','Confirmed integrity problems');
  if(!(q.integrityProblems||[]).length)empty(integrity,'No deterministic failures in the analyzed observations.');
  for(const f of q.integrityProblems||[]){const card=node('article','finding');card.append(node('div','title',`${f.gameKey} \u00b7 ${fmt(f.observedAt)}`),node('div','detail',valueText(f.codes||f.detail)),evidenceDetails('Integrity evidence',f));integrity.append(card);}
  if(q.historicalFindingReview?.length){
    const history=qualitySection(el,'quality-historical-review','Historical findings reconsidered');
    history.append(node('p','evidence','Original findings remain in the export. These are updated interpretations, not changes to historical records.'));
    for(const r of q.historicalFindingReview){const card=node('article','finding');card.append(node('div','title',`${r.gameKey} \u00b7 ${r.reasonCode}`),node('div','detail',`${label(r.classification)}. ${r.reason||r.detail||''}`),evidenceDetails('Supporting evidence',r));history.append(card);}
  }
  const priorities=qualitySection(el,'quality-priority','High-priority scoring-quality concerns');
  const priority=recommendations.filter(r=>['confirmed_defect','strong_calibration_concern','calibration_hypothesis'].includes(r.classification));
  if(priority.length)priority.slice(0,10).forEach(r=>priorities.append(qualityCard(r)));else empty(priorities,'No supported high-priority scoring change. Evidence gaps and investigation tasks remain below.');
  for(const [id,title,items] of [['quality-over','Potentially overrated games',q.topOverrated||[]],['quality-under','Potentially underrated games',q.topUnderrated||[]]]){
    const section=qualitySection(el,id,title);if(items.length)items.forEach(r=>section.append(qualityCard(r)));else empty(section,'No supported candidates; coverage may be incomplete.');
  }
  const eventSection=qualitySection(el,'quality-events','Significant events and rating responses');
  const eventPriority=e=>Number.isFinite(e.eventTimestamp??e.event?.occurredAt)?Number.isFinite(e.change??e.delta??e.ratingChange)?0:1:2;
  const responses=games.flatMap(g=>(g.eventResponses||[]).map(e=>({...e,gameKey:g.gameKey,modelVersion:g.modelVersion,matchup:g.matchup}))).sort((a,b)=>eventPriority(a)-eventPriority(b));
  eventSection.append(node('p','evidence','Observed response intervals are associations. Provider publication latency, polling delay and scoring behavior stay separate; unavailable timestamps remain unavailable.'));
  if(!responses.length)empty(eventSection,'No usable event-response measurements in retained data.');
  for(const e of responses.slice(0,40)){
    const card=node('article','game');card.append(node('div','title',e.matchup||e.gameKey),node('div','detail',`${e.event?.text||e.event?.type||e.eventType||e.event?.id||'State transition'}: ${valueText(e.beforeRating??e.ratingBefore??e.baseline)} to ${valueText(e.afterRating??e.ratingAfter??e.rating)}; change ${valueText(e.change??e.delta??e.ratingChange)}.`),node('div','evidence',`${e.event?.retrospective?'Retrospective later-cache association; historical availability unknown. ':''}${label(e.status)}. Direction: ${valueText(e.directionAssessment??{expected:e.expectedDirection,consistent:e.directionConsistent,magnitude:e.magnitudeAssessment})}.`),observationLink(e.gameKey,e.modelVersion),evidenceDetails('Timing, uncertainty and concurrent events',e));eventSection.append(card);
  }
  if(responses.length>40)eventSection.append(node('p','evidence',`Showing 40 of ${responses.length} responses. Full details remain in the ZIP export and game trajectories.`));
  const volatility=qualitySection(el,'quality-volatility','Rating volatility and discontinuities');
  const moves=games.filter(g=>g.volatility?.majorChanges?.length);
  if(!moves.length)empty(volatility,'No major changes supported by comparable retained observations.');
  for(const g of moves){const card=node('article','game');card.append(node('div','title',`${g.matchup||g.gameKey} \u00b7 ${g.modelVersion}`),node('div','detail',`Maximum observed change ${g.volatility.maxAbsoluteChange}; ${g.volatility.majorChanges.length} major changes.`),observationLink(g.gameKey,g.modelVersion),evidenceDetails('Component changes and comeback evidence',{volatility:g.volatility,comebackAnalysis:g.comebackAnalysis}));volatility.append(card);}
  const alerts=qualitySection(el,'quality-alerts','Alert usefulness');
  alerts.append(node('p','evidence','First observed crossings can be later than the true crossing. Remaining game time is not wall-clock viewing time. Episodes and sampled duration exclude uncertain gaps. Notification behavior is unchanged.'));
  if(!games.some(g=>g.thresholds?.length))empty(alerts,'Threshold trajectories are unavailable.');
  for(const g of games.filter(g=>g.thresholds?.length)){
    const t=g.thresholds.find(t=>t.threshold===85),card=node('article','game');
    card.append(node('div','title',`${g.matchup||g.gameKey} \u00b7 ${g.modelVersion}`),node('div','detail',t?.firstObservedAt?`First observed 85+: ${fmt(t.firstObservedAt)}; game time remaining ${valueText(t.gameTimeRemainingSeconds)} seconds; ${t.episodeCount} episode(s).`:'No 85+ crossing observed in this cohort.'),node('div','evidence',`Usefulness: ${valueText(t?.firstCrossingUsefulness)}. Sampled time 85+: ${valueText(t?.estimatedSecondsAbove)} seconds.`),observationLink(g.gameKey,g.modelVersion),evidenceDetails('70 / 80 / 85 / 90 / 95 crossings and uncertainty',g.thresholds));alerts.append(card);
  }
  const calibration=qualitySection(el,'quality-calibration','Sport-specific calibration and model-version comparisons');
  calibration.append(node('p','evidence','Each model version has its own denominator and distribution. Different cohorts are observational, not a controlled release comparison. Soccer is hidden; tennis calibration is limited to verified Grand Slams.'));
  for(const cohort of q.modelCohorts||[]){
    calibration.append(node('h4',null,`Calibration \u00b7 ${cohort.modelVersion}`));
    for(const s of cohort.sports||[]){
      if(s.sport==='Soccer')continue;
      const card=node('article','game');card.append(node('div','title',s.displaySport||(s.sport==='NCAAB'?'NCAAM':s.sport)));
      card.append(node('div','detail',`${s.gamesWithRating??0} games with ratings \u00b7 reaching 70/80/85/90/95: ${Object.values(s.percentageReaching||{}).map(v=>v===null?'unknown':Number(v).toFixed(1)+'%').join(' / ')} \u00b7 peak median ${s.peakDistribution?.median??'unknown'} \u00b7 seconds >85: ${s.estimatedSecondsAbove85??'unknown'}`));
      card.append(node('div','evidence',`${(s.significantNeverReached85||[]).length} significant games never observed at 85. ${s.comparability85||'Equivalence across sports is unvalidated.'}`),evidenceDetails('Calibration evidence and limitations',s));calibration.append(card);
    }
  }
  if(!(q.modelCohorts||[]).length)empty(calibration,'No retained live observations. All sport calibration metrics are unavailable.');
  const evidence=qualitySection(el,'quality-evidence','Evidence coverage and limitations');
  evidence.append(node('p','evidence','Complete, partial, invalid and unavailable describe the retained evidence. Legacy snapshots are not relabeled as complete. Refresh start and acquisition timestamps do not establish provider publication latency.'),evidenceDetails('Coverage counts',q.evidenceCoverage||{unavailable:'No structured coverage report'}),evidenceDetails('Missing evidence by game and model',q.missingEvidence||[]));
  const retrospectiveResponses=responses.filter(e=>e.event?.retrospective);
  evidence.append(node('p','evidence',`Retrospective event associations: ${retrospectiveResponses.length}; with provider event wallclock: ${retrospectiveResponses.filter(e=>Number.isFinite(e.eventTimestamp??e.event?.occurredAt)).length}. Missing-timestamp counts describe original historical observations. Later-cache events are separate evidence and never backfill those observations.`));
  const investigation=qualitySection(el,'quality-recommendations','Recommended investigations');
  for(const f of q.qualitativeFindings||[])investigation.append(evidenceDetails(`${f.gameKey} · ${f.reasonCode} · ${f.classification}`,f));
  if(!recommendations.length)empty(investigation,'Additional event and scoring-component evidence is required before a supported scoring change can be recommended.');
  for(const r of recommendations)investigation.append(qualityCard(r));
  const trajectories=qualitySection(el,'quality-trajectories','Supporting game observations');
  trajectories.append(node('p','evidence','Each identifier maps to retained observations in findings.json. The dashboard previews up to 80 trajectory points per game/model; the export contains every retained point within the reported read limits.'));
  for(const g of games){
    const details=node('details','game');details.id=trajectoryId(g.gameKey,g.modelVersion);
    details.append(node('summary',null,`${g.matchup||g.gameKey} \u00b7 ${g.modelVersion}`),node('p','evidence',g.gameKey),node('pre','evidence',JSON.stringify({ratingContexts:g.ratingContexts,trajectory:(g.ratingTrajectory||[]).slice(0,80),watchability:g.watchability,comebackAnalysis:g.comebackAnalysis,eventResponses:g.eventResponses},null,2)));trajectories.append(details);
  }
  evidence.append(evidenceDetails('Methodology and analysis limits',{methodology:q.methodology,limitations:q.limitations,simultaneousPriorityConflicts:q.simultaneousPriorityConflicts}));
}
let qualityGeneration=0;
function invalidateQuality(){qualityGeneration++;clear($('quality'));$('quality-status').textContent='Load quality for the selected date.';}
async function loadQuality(){
  const generation=++qualityGeneration,date=$('day').value||dayNY();
  $('quality-load').disabled=true;$('quality-status').textContent='Analyzing retained live observations…';
  try{
    const response=await fetch(`/quality?date=${encodeURIComponent(date)}`,{credentials:'same-origin',cache:'no-store'});
    if(response.status===401){location.href='/dashboard';return}if(!response.ok)throw Error(`Quality HTTP ${response.status}`);
    const data=await response.json();if(generation!==qualityGeneration)return;
    const q=data.scoringQuality,el=$('quality');clear(el);
    $('quality-status').textContent=`${data.dateNewYork} · ${q.modelCohorts.length} scoring model(s). Hypotheses require review; an equivalent 85 across sports is not established. Duration is a sampled estimate. Missing evidence is not uneventful play.`;
    renderQuality(q,el,date);
  }catch(error){if(generation===qualityGeneration)$('quality-status').textContent=String(error);}
  finally{$('quality-load').disabled=false;}
}
$('quality-load').addEventListener('click',loadQuality);
$('refresh').addEventListener('click',refresh);$('day').addEventListener('change',()=>{invalidateQuality();refresh();});
$('export').addEventListener('click',()=>{const date=$('day').value || dayNY();location.href=`/export?date=${encodeURIComponent(date)}`});
init();setInterval(refresh,60000);
